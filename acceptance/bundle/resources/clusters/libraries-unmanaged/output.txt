=== The redeploy installs urllib3 and restarts to apply it, but leaves the out-of-band 'six' alone (no uninstall)
>>> print_requests.py //libraries/uninstall //libraries/install //clusters/restart
{
  "method": "POST",
  "path": "/api/2.0/libraries/install",
  "body": {
    "cluster_id": "[MYCLUSTER_ID]",
    "libraries": [
      {
        "pypi": {
          "package": "requests"
        }
      },
      {
        "pypi": {
          "package": "urllib3"
        }
      }
    ]
  }
}
{
  "method": "POST",
  "path": "/api/2.1/clusters/restart",
  "body": {
    "cluster_id": "[MYCLUSTER_ID]"
  }
}
